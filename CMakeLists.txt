cmake_minimum_required(VERSION 3.16)
set(PROJECT_NAME LayaNative3.0)
project(${PROJECT_NAME})
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)


set(CONCH_ROOT_DIR ${CMAKE_CURRENT_LIST_DIR})
include(${CONCH_ROOT_DIR}/cmake/preprocessor.cmake)

if(ANDROID)
	add_definitions(-DJS_V8)
	option(CONCH_USE_CURL_DOWNLOAD "Use CURL" OFF)
	option(USE_JS_V8_DEBUGGER "Use v8 debugger" ON)
elseif(IOS)
	add_definitions(-DJS_V8)
	option(CONCH_USE_CURL_DOWNLOAD "Use CURL" OFF)
	option(USE_JS_V8_DEBUGGER "Use v8 debugger" OFF)
elseif(MSVC)
	add_definitions(-DJS_V8)
	option(CONCH_USE_CURL_DOWNLOAD "Use CURL" ON)
	option(USE_JS_V8_DEBUGGER "Use v8 debugger" ON)
elseif(OHOS)
	option(CONCH_USE_CURL_DOWNLOAD "Use CURL" ON)
	option(USE_JS_V8_DEBUGGER "Use v8 debugger" ON)
	find_library(EGL-lib EGL)
	find_library(GLESv3-lib GLESv3)
	find_library(hilog-lib hilog_ndk.z)
	find_library(libace-lib ace_ndk.z)
	find_library(libnapi-lib ace_napi.z)
	find_library(libuv-lib uv)
	find_library(rawfile-lib rawfile.z)
	find_library(Drawing-lib native_drawing)
	add_definitions(-DOHOS)
	add_definitions(-DJS_V8)
	add_definitions(-fexceptions)
	#add_definitions(-DIN_LIBRARY)
	#add_definitions(-D_GLIBCXX_PERMIT_BACKWARD_HASH)
	add_definitions(-DOHOS_ARM_NEON=TRUE)
	add_definitions(-DOHOS_STL=c++_shared)
	add_definitions(-DOHOS_TOOLCHAIN=clang)
	add_definitions(-fno-rtti)
	add_definitions(-DAL_ALEXT_PROTOTYPES)
elseif(LINUX)
	add_definitions(-DJS_V8)
	add_definitions(-D__LINUX__)
	option(CONCH_USE_CURL_DOWNLOAD "Use CURL" ON)
endif()

if(USE_JS_V8_DEBUGGER)
	set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -DJS_V8_DEBUGGER")
endif()

if(CONCH_USE_CURL_DOWNLOAD)
	add_definitions(-DCURL_DOWNLOAD)
endif()

if(MSVC)
 	add_compile_definitions(UNICODE _UNICODE)
	set_property(DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR} PROPERTY VS_STARTUP_PROJECT launcher)
	# set(CMAKE_C_FLAGS "${CMAKE_C_FLAGS}")
	set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} /std:c++latest /W0 /Zc:__cplusplus")
	set(CMAKE_CXX_FLAGS_DEBUG "${CMAKE_CXX_FLAGS_DEBUG} -D_DEBUG -std=c++17 -O0")
	set(CMAKE_CXX_FLAGS_RELEASE "${CMAKE_CXX_FLAGS_RELEASE} -DNDEBUG -std=c++17")
	file(GLOB WIN_DLLS 
		${CMAKE_CURRENT_SOURCE_DIR}/third_party/v8/lib/win64/*.dll 
		${CMAKE_CURRENT_SOURCE_DIR}/third_party/angle/lib/win64/*.dll
		${CMAKE_CURRENT_SOURCE_DIR}/third_party/openal/lib/win64/*.dll
	)
else()
	set(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} -fPIC")
	set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -fno-rtti -std=c++17")
	set(CMAKE_CXX_FLAGS_DEBUG "${CMAKE_CXX_FLAGS_DEBUG} -D_DEBUG -std=c++17 -O0")
	set(CMAKE_CXX_FLAGS_RELEASE "${CMAKE_CXX_FLAGS_RELEASE} -DNDEBUG -std=c++17")
endif()

# 设置可执行文件的输出目录
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)

if(NOT BUILDING_CONCH_SHARED AND NOT BUILDING_CONCH_STATIC)
   add_definitions(-DBUILDING_CONCH_FROME_SOURCE)
   if(NOT LINUX AND NOT OHOS)
		add_subdirectory(template)
   endif()
endif()

add_subdirectory(modules/utils)
add_subdirectory(modules/binder)
add_subdirectory(conch)

add_subdirectory(third_party/iniparser iniparser)
add_subdirectory(third_party/zlib zlib)
add_subdirectory(third_party/zip zip)
add_subdirectory(third_party/websockets websockets)
add_subdirectory(third_party/openssl openssl)
add_subdirectory(third_party/png png)
add_subdirectory(third_party/rapidxml rapidxml)
add_subdirectory(third_party/ghc ghc)
add_subdirectory(third_party/jpeg-turbo jpeg-turbo)
if(NOT LINUX)
add_subdirectory(third_party/mpg123 mpg123)
endif()
if(OHOS)
add_subdirectory(third_party/aki aki_jsbind)
endif()
add_subdirectory(third_party/openal openal)
add_subdirectory(third_party/ogg ogg)
add_subdirectory(third_party/freetype freetype)
add_subdirectory(third_party/curl curl)
if(OHOS)
add_subdirectory(third_party/v8_9.1 v8)
else()
add_subdirectory(third_party/v8 v8)
endif()
add_subdirectory(third_party/angle angle)
add_subdirectory(third_party/bullet bullet)
add_subdirectory(third_party/physx physx)
add_subdirectory(third_party/sdl sdl)
add_subdirectory(third_party/benchmark benchmark)
